import { spawn } from 'node:child_process'
import { createRequire } from 'node:module'
import { access, chmod, lstat, mkdtemp, mkdir, open, readFile, rename, rm } from 'node:fs/promises'
import { constants } from 'node:fs'
import { homedir, tmpdir } from 'node:os'
import { dirname, join, resolve } from 'node:path'
import { fileURLToPath } from 'node:url'

const applicationName = '@@APPLICATION_NAME@@'
const foregroundCommands = new Set(['install', 'list'])
const electronDiagnosticPattern = /^\[\d+:\d+\/\d+\.\d+:(?:ERROR|WARNING):/
const appRoot = resolve(dirname(fileURLToPath(import.meta.url)), '..')
const mainProcessRequire = createRequire(join(appRoot, 'packages', 'main-process', 'package.json'))

const isVersionRequest = (args) => args.includes('--version') || args.includes('-v')

const isForegroundRequest = (args) => {
  if (args.some((arg) => ['--built-in-self-test', '--help', '--status', '--verbose', '--wait', '--web'].includes(arg))) {
    return true
  }
  return args.some((arg) => foregroundCommands.has(arg))
}

const shellQuote = (value) => `'${value.replaceAll("'", "'\\''")}'`

const parseElectronVersion = (args) => {
  const filteredArgs = []
  let electronVersion
  for (let index = 0; index < args.length; index++) {
    const arg = args[index]
    if (arg === '--electron-version' || arg.startsWith('--electron-version=')) {
      const value = arg === '--electron-version' ? args[++index] : arg.slice('--electron-version='.length)
      if (!value || value.startsWith('--') || !/^v?\d+\.\d+\.\d+(?:-[\w.-]+)?$/.test(value)) {
        throw new Error('--electron-version requires a valid Electron version (for example, --electron-version=44.0.0)')
      }
      if (!electronVersion) {
        electronVersion = value.replace(/^v/, '')
      }
      continue
    }
    filteredArgs.push(arg)
  }
  return { electronVersion, args: filteredArgs }
}

const getElectronExecutablePath = (electronDir) => {
  if (process.platform === 'darwin') {
    return join(electronDir, 'Electron.app', 'Contents', 'MacOS', 'Electron')
  }
  if (process.platform === 'win32') {
    return join(electronDir, 'electron.exe')
  }
  return join(electronDir, 'electron')
}

const runPermissionCommand = async (command, args) => {
  const isRoot = typeof process.getuid === 'function' && process.getuid() === 0
  const executable = isRoot ? command : 'sudo'
  const commandArgs = isRoot ? args : [command, ...args]
  const child = spawn(executable, commandArgs, { stdio: 'inherit' })
  const { promise, resolve, reject } = Promise.withResolvers()
  child.once('error', reject)
  child.once('exit', (code) => {
    if (code === 0) {
      resolve()
    } else {
      reject(new Error(`${executable} exited with code ${code ?? 1}`))
    }
  })
  await promise
}

const ensureLinuxSandbox = async (electronDir) => {
  if (process.platform !== 'linux') {
    return
  }

  const sandboxPath = join(electronDir, 'chrome-sandbox')
  let runtimeDirectory
  try {
    runtimeDirectory = await open(electronDir, constants.O_RDONLY | constants.O_DIRECTORY | constants.O_NOFOLLOW)
  } catch (error) {
    throw new Error(`Electron runtime directory must be a real directory at ${electronDir}: ${error instanceof Error ? error.message : error}`)
  }
  const permissionPath = `/proc/${process.pid}/fd/${runtimeDirectory.fd}/chrome-sandbox`
  try {
    const sandbox = await lstat(permissionPath).catch(() => {
      throw new Error(`Electron sandbox helper is missing at ${sandboxPath}`)
    })
    if (!sandbox.isFile()) {
      throw new Error(`Electron sandbox helper must be a regular file at ${sandboxPath}`)
    }
    if (sandbox.uid === 0 && (sandbox.mode & 0o7777) === 0o4755) {
      return
    }

    try {
      await runPermissionCommand('chown', ['root:root', '--no-dereference', '--', permissionPath])
      await runPermissionCommand('chmod', ['4755', '--no-dereference', '--', permissionPath])
      const repairedSandbox = await lstat(permissionPath)
      if (!repairedSandbox.isFile() || repairedSandbox.uid !== 0 || (repairedSandbox.mode & 0o7777) !== 0o4755) {
        throw new Error('permission verification failed')
      }
    } catch (error) {
      throw new Error(
        `Unable to prepare Electron sandbox helper at ${sandboxPath}; run sudo chown root:root -- ${shellQuote(sandboxPath)} and sudo chmod 4755 -- ${shellQuote(sandboxPath)}: ${error instanceof Error ? error.message : error}`,
        { cause: error },
      )
    }
  } finally {
    await runtimeDirectory.close()
  }
}

const ensureElectronRuntime = async (electronVersion) => {
  const cacheHome = process.env.XDG_CACHE_HOME || join(homedir(), '.cache')
  const cacheRoot = join(cacheHome, applicationName, 'electron')
  const cachePath = join(cacheRoot, `${electronVersion}-${process.platform}-${process.arch}`)
  const executablePath = getElectronExecutablePath(cachePath)
  let runtimeCached = true
  try {
    await access(executablePath, constants.X_OK)
  } catch {
    runtimeCached = false
  }
  if (runtimeCached) {
    await ensureLinuxSandbox(cachePath)
    return executablePath
  }

  await mkdir(cacheRoot, { recursive: true })
  const stagingPath = await mkdtemp(join(cacheRoot, `.download-${electronVersion}-`))
  try {
    const electronGetPath = mainProcessRequire.resolve('@electron/get')
    const extractZipPath = mainProcessRequire.resolve('@electron-internal/extract-zip')
    const electronGet = await import(electronGetPath)
    const extractZipModule = await import(extractZipPath)
    console.log(`Downloading Electron ${electronVersion}...`)
    const zipPath = await electronGet.downloadArtifact({
      version: electronVersion,
      platform: process.platform,
      artifactName: 'electron',
      arch: process.arch,
    })
    await extractZipModule.default(zipPath, { dir: stagingPath })
    const stagedExecutablePath = getElectronExecutablePath(stagingPath)
    await access(stagedExecutablePath, constants.X_OK).catch(async () => {
      await chmod(stagedExecutablePath, 0o755)
      await access(stagedExecutablePath, constants.X_OK)
    })
    await ensureLinuxSandbox(stagingPath)
    try {
      await rename(stagingPath, cachePath)
    } catch (error) {
      try {
        await access(executablePath, constants.X_OK)
      } catch {
        throw error
      }
      await ensureLinuxSandbox(cachePath)
    }
    return executablePath
  } catch (error) {
    throw new Error(`Failed to prepare Electron ${electronVersion}: ${error instanceof Error ? error.message : error}`, { cause: error })
  } finally {
    await rm(stagingPath, { recursive: true, force: true })
  }
}

const getTransientLaunchOptions = async (args) => {
  if (!args.includes('--transient')) {
    return { args, env: {} }
  }

  const root = await mkdtemp(join(tmpdir(), `${applicationName}-`))
  const configHome = join(root, 'config')
  const dataHome = join(root, 'data')
  const cacheHome = join(root, 'cache')
  const stateHome = join(root, 'state')
  const userDataDir = join(configHome, applicationName, 'electron')
  const env = {
    XDG_CACHE_HOME: cacheHome,
    XDG_CONFIG_HOME: configHome,
    XDG_DATA_HOME: dataHome,
    XDG_STATE_HOME: stateHome,
  }
  const relaunchArguments = [
    `XDG_CONFIG_HOME=${shellQuote(configHome)}`,
    `XDG_DATA_HOME=${shellQuote(dataHome)}`,
    `XDG_CACHE_HOME=${shellQuote(cacheHome)}`,
    `XDG_STATE_HOME=${shellQuote(stateHome)}`,
    applicationName,
    `--user-data-dir=${shellQuote(userDataDir)}`,
  ]
  console.log(`State is temporarily stored. Relaunch this state with: ${relaunchArguments.join(' ')}`)
  return {
    args: [...args, `--user-data-dir=${userDataDir}`],
    env,
  }
}

const forwardElectronStderr = (stream) => {
  let pending = ''
  stream.setEncoding('utf8')
  stream.on('data', (chunk) => {
    pending += chunk
    const lines = pending.split('\n')
    pending = lines.pop() || ''
    for (const line of lines) {
      if (!electronDiagnosticPattern.test(line)) {
        process.stderr.write(`${line}\n`)
      }
    }
  })
  stream.on('end', () => {
    if (pending && !electronDiagnosticPattern.test(pending)) {
      process.stderr.write(pending)
    }
  })
}

const launchElectron = async (args, envOverrides) => {
  const env = { ...process.env, ...envOverrides }
  delete env.ELECTRON_RUN_AS_NODE

  const foreground = isForegroundRequest(args)
  const verbose = args.includes('--verbose')
  const parsedArgs = parseElectronVersion(args)
  const executablePath = parsedArgs.electronVersion ? await ensureElectronRuntime(parsedArgs.electronVersion) : process.execPath
  const launchArgs = parsedArgs.electronVersion ? [appRoot, ...parsedArgs.args] : args
  const child = spawn(executablePath, launchArgs, {
    detached: true,
    env,
    stdio: foreground ? ['inherit', 'inherit', 'pipe'] : 'ignore',
  })

  if (!foreground) {
    const { promise, resolve, reject } = Promise.withResolvers()
    child.once('error', reject)
    child.once('spawn', resolve)
    await promise
    child.unref()
    return 0
  }

  if (verbose) {
    child.stderr.pipe(process.stderr)
  } else {
    forwardElectronStderr(child.stderr)
  }

  const forwardSigInt = () => child.kill('SIGINT')
  const forwardSigTerm = () => child.kill('SIGTERM')
  process.on('SIGINT', forwardSigInt)
  process.on('SIGTERM', forwardSigTerm)
  try {
    const { promise, resolve, reject } = Promise.withResolvers()
    child.once('error', reject)
    child.once('exit', (code) => resolve(code ?? 1))
    return await promise
  } finally {
    process.off('SIGINT', forwardSigInt)
    process.off('SIGTERM', forwardSigTerm)
  }
}

const main = async () => {
  const args = process.argv.slice(2)
  if (isVersionRequest(args)) {
    const packageJson = JSON.parse(await readFile(new URL('../package.json', import.meta.url), 'utf8'))
    console.log(packageJson.version)
    return 0
  }
  const launchOptions = await getTransientLaunchOptions(args)
  return launchElectron(launchOptions.args, launchOptions.env)
}

try {
  process.exitCode = await main()
} catch (error) {
  console.error(error instanceof Error ? error.message : error)
  process.exitCode = 1
}
